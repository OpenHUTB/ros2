#!/usr/bin/env python3
import rospy
from geometry_msgs.msg import Twist
from turtlesim.msg import Pose
import math
import time

# 目标位置和角度（画一个正方形）
target_x = 5.5
target_y = 5.5
target_theta = 0.0

# 当前位姿
current_x = 0.0
current_y = 0.0
current_theta = 0.0

# 标志位：是否已经到达过目标点（简化逻辑，直接走正方形）
step = 0

def pose_callback(msg):
    global current_x, current_y, current_theta
    current_x = msg.x
    current_y = msg.y
    current_theta = msg.theta

def move_square():
    global step
    rospy.init_node('square_move', anonymous=True)
    
    # 订阅小海龟的位置信息
    rospy.Subscriber('/turtle1/pose', Pose, pose_callback)
    
    # 发布速度控制指令
    pub = rospy.Publisher('/turtle1/cmd_vel', Twist, queue_size=10)
    
    # 设定发布频率
    rate = rospy.Rate(10) # 10Hz
    
    # 等待一下，确保连接成功
    time.sleep(2)
    
    # 正方形的边长（单位：米/像素）
    side_length = 2.0
    # 转角速度（大概1.57弧度，对应90度）
    turn_angle = math.pi / 2
    
    # 让海龟走正方形，一共走4条边
    for i in range(4):
        rospy.loginfo(f"正在画第 {i+1} 条边...")
        # 1. 前进
        move_cmd = Twist()
        move_cmd.linear.x = 1.0  # 线速度
        move_cmd.angular.z = 0.0 # 角速度
        
        # 计算前进需要的时间：距离 / 速度 = 2.0 / 1.0 = 2秒
        move_time = side_length / 1.0
        
        # 开始前进
        start_time = time.time()
        while time.time() - start_time < move_time:
            pub.publish(move_cmd)
            rate.sleep()
            
        # 停止一小会
        move_cmd.linear.x = 0.0
        pub.publish(move_cmd)
        time.sleep(0.5)
        
        # 2. 转弯
        rospy.loginfo(f"正在转第 {i+1} 个弯...")
        turn_cmd = Twist()
        turn_cmd.linear.x = 0.0
        turn_cmd.angular.z = 1.0 # 角速度（弧度/秒）
        
        # 计算转弯需要的时间：角度 / 角速度 = 1.57 / 1.0 ≈ 1.57秒
        turn_time = turn_angle / 1.0
        
        start_time = time.time()
        while time.time() - start_time < turn_time:
            pub.publish(turn_cmd)
            rate.sleep()
            
        # 停止一小会
        turn_cmd.angular.z = 0.0
        pub.publish(turn_cmd)
        time.sleep(0.5)
        
    rospy.loginfo("正方形绘制完成！")
    # 停止海龟
    stop_cmd = Twist()
    pub.publish(stop_cmd)

if __name__ == '__main__':
    try:
        move_square()
    except rospy.ROSInterruptException:
        pass
